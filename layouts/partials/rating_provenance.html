{{/* Rating provenance — the ONE rule for whether a rating/count may be presented or emitted as an Amazon observation. A rating is provenanced only when BOTH exist in frontmatter: ratings_observed: integer >= 1 (an actually observed count) observed_at: date, not older than 90 days No derivation, no defaults, no fallback estimate. Absent either field the page may still show our own verdict, but it must not call the number an Amazon rating and must not emit AggregateRating. Returns dict: provenanced (bool) · count (int) · at (string) */}} {{- $count := 0 -}} {{- $at := "" -}} {{- with .Params.ratings_observed -}}{{- $count = int . -}}{{- end -}} {{- with .Params.observed_at -}}{{- $at = . -}}{{- end -}} {{- $provenanced := false -}} {{- if and (ge $count 1) $at -}} {{- $when := time.AsTime $at -}} {{- if ge $when (now.AddDate 0 0 -90) -}} {{- $provenanced = true -}} {{- end -}} {{- end -}} {{- return (dict "provenanced" $provenanced "count" $count "at" $at) -}}
