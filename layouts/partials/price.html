{{- /* Single source of truth for "does this page hold a real, quotable price?". Returns a dict: isNumeric : bool — true only for a bare numeric value value : float — 0 when not numeric raw : string — the raw param as text currency : string — from the marketplace, never hardcoded country : string — merchant return-policy country seller : string — marketplace seller name Two independent checks (homepage ranking, schema.org offers) would drift. Both read price through this partial instead. Text values such as "Check price on Amazon" or "N/A" must never be treated as a price. Call: {{ $p := partial "price.html" . }} */ -}} {{- $raw := printf "%v" (.Params.price | default "") -}} {{- $isNumeric := false -}} {{- $value := 0.0 -}} {{- if findRE "^[0-9]+(?:\\.[0-9]+)?$" $raw -}} {{- $isNumeric = true -}} {{- $value = (float $raw) -}} {{- end -}} {{- $url := .Params.amazon_url | default "" -}} {{- $currency := "USD" -}} {{- $country := "US" -}} {{- $seller := "Amazon.com" -}} {{- if in $url "amazon.sg" -}} {{- $currency = "SGD" -}}{{- $country = "SG" -}}{{- $seller = "Amazon.sg" -}} {{- else if in $url "amazon.co.uk" -}} {{- $currency = "GBP" -}}{{- $country = "GB" -}}{{- $seller = "Amazon.co.uk" -}} {{- else if in $url "amazon.ca" -}} {{- $currency = "CAD" -}}{{- $country = "CA" -}}{{- $seller = "Amazon.ca" -}} {{- else if in $url "amazon.com.au" -}} {{- $currency = "AUD" -}}{{- $country = "AU" -}}{{- $seller = "Amazon.com.au" -}} {{- end -}} {{- return (dict "isNumeric" $isNumeric "value" $value "raw" $raw "currency" $currency "country" $country "seller" $seller) -}}
